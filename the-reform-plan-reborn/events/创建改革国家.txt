	hide_window = yes

	is_triggered_only = yes

	immediate = {
		country_event = {
			id = tt_reform_plan.1
		}
	}
}

#开局创建tt_reform_plan国家类型
country_event = {
	id = tt_reform_plan.1
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		create_country = {
			type = tt_reform_plan
			name = "NAME_tt_reform_plan_country"
			effect = {
				set_country_flag = tt_reform_plan@root
				change_country_flag = {
					colors = {
						"red"
						"orange"
					}
					icon = {
						category = "human"
						file = "flag_human_1.dds"
					}
					background= {
						category = "backgrounds"
						file = "vertical.dds"
					}
				}
			}
		}
	}
}

#检查是否为tt_reform_plan国家类型，并与其联系
country_event = {
	id = tt_reform_plan.2
	hide_window = yes

	is_triggered_only = yes

	trigger = {
		FROM = {
			is_country_type = tt_reform_plan
		}
		NOT = { has_country_flag = flag_the_reform_plan }
	}
	immediate = {
		country_event = {
			id = tt_reform_plan.3
		}
		set_country_flag = flag_the_reform_plan
	}
}

#与tt_reform_plan国家类型的联系界面（改革计划主界面）
country_event = {
	id = tt_reform_plan.3
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		if = {
			limit = {
				has_ascension_perk = ap_tt_reform_plan_2
			}
			country_event = { id = tt_reform_plan.6 }
		}
		else = {
			country_event = { id = tt_reform_plan.5 }
		}
	}
}

#点了“突破计划”后的隐藏事件，并与tt_reform_plan国家类型建立无通知的联系
country_event = {
	id = tt_reform_plan.4
	hide_window = yes

	is_triggered_only = yes

	immediate = {
		random_country = {
			limit = { is_country_type = tt_reform_plan }
			save_event_target_as = the_tt_reform_plan
		}
		establish_communications_no_message = event_target:the_tt_reform_plan #建立无通知的联系
		country_event = { id = tt_reform_plan.3 } #但会瞬间转移到与tt_breakthrough国家类型的通讯界面
	}
}

#判断改革计划应该打开哪一个面板：主面板
